# AutoInsight Frontend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a local web UI to AutoInsight — FastAPI backend with SSE progress streaming and a single-page dark-theme frontend with a node-chain pipeline visualizer.

**Architecture:** FastAPI serves `app/static/index.html` and three API endpoints. `POST /analyze` accepts a CSV upload + form fields, saves the file to a temp dir, starts graph execution in a background thread, and returns a `task_id`. `GET /progress/{task_id}` streams SSE events per node. `GET /report/{task_id}` returns the generated Markdown as a downloadable file. The graph runs unchanged via `graph.stream()`, which yields per-node state updates; a `queue.Queue` bridges the sync thread and the async SSE generator.

**Tech Stack:** FastAPI, Uvicorn, python-multipart, LangGraph `graph.stream()`, SSE, vanilla HTML/CSS/JS

```python
@app.post("/analyze")
async def analyze(
    file: UploadFile = File(...),
    prompt: str = Form(...),
    target: str = Form(default=""),
    level: str = Form(default="general"),
    tune: str = Form(default="false"),
):
    if not file.filename.lower().endswith(".csv"):
        raise HTTPException(status_code=400, detail="Only CSV files are supported.")

    task_id = str(uuid.uuid4())

    # Save upload to temp dir
    tmp_dir = tempfile.mkdtemp(prefix=f"autoinsight_{task_id}_")
    TASK_TMPDIRS[task_id] = tmp_dir
    file_path = os.path.join(tmp_dir, file.filename)
    with open(file_path, "wb") as f:
        f.write(await file.read())

    # Init progress queue
    TASK_QUEUES[task_id] = queue.Queue()

    # Build initial state
    import sys
    sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
    tune_bool = tune.lower() in ("true", "1", "yes")

    initial_state = {
        "user_query":    prompt,
        "file_path":     file_path,
        "target_column": target,
        "user_level":    level,
        "tune":          tune_bool,
        "logs":          [],
        "schema":               {},
        "quality_issues":       [],
        "task_category":        "",
        "task_type":            "",
        "user_intent_summary":  "",
        "selected_algorithms":  [],
        "reasoning":            "",
        "model_results":        {},
        "metrics":              {},
        "best_model":           "",
        "charts":               [],
        "eda_summary":          {},
        "modeling_hints":       {},
        "feature_names":        [],
        "X_train": None, "X_test": None,
        "y_train": None, "y_test": None,
        "report_path":          "",
    }

    _start_analysis(task_id, initial_state)
    return {"task_id": task_id}


def _start_analysis(task_id: str, initial_state: dict) -> None:
    """Launch graph execution in a background thread."""
    t = threading.Thread(
        target=_run_graph,
        args=(task_id, initial_state),
        daemon=True,
    )
    t.start()


def _run_graph(task_id: str, initial_state: dict) -> None:
    q = TASK_QUEUES[task_id]
    t0 = time.time()
    try:
        from app.graph import build_graph
        graph = build_graph()

        final_state = dict(initial_state)
        for chunk in graph.stream(initial_state):
            for node_name, state_update in chunk.items():
                if isinstance(state_update, dict):
                    final_state.update(state_update)
                q.put(json.dumps({"node": node_name, "status": "done"}))

        elapsed = int(time.time() - t0)
        TASK_RESULTS[task_id] = final_state
        q.put(json.dumps({
            "event":      "complete",
            "task_type":  final_state.get("task_type", ""),
            "best_model": final_state.get("best_model", "") or "N/A",
            "elapsed":    elapsed,
        }))
    except Exception as exc:
        q.put(json.dumps({"event": "error", "message": str(exc)}))
    finally:
        q.put(None)  # sentinel — tells SSE generator to close
```

@app.get("/progress/{task_id}")
async def progress(task_id: str):
    if task_id not in TASK_QUEUES:
        raise HTTPException(status_code=404, detail="Task not found.")

    async def event_stream():
        q = TASK_QUEUES[task_id]
        loop = asyncio.get_event_loop()
        while True:
            # Blocking q.get() offloaded to thread pool so event loop stays free
            item = await loop.run_in_executor(None, q.get)
            if item is None:           # sentinel → analysis finished
                del TASK_QUEUES[task_id]
                break
            yield f"data: {item}\n\n"

    return StreamingResponse(event_stream(), media_type="text/event-stream")
```

@app.get("/report/{task_id}")
def report(task_id: str):
    if task_id not in TASK_RESULTS:
        raise HTTPException(status_code=404, detail="Task not found or not yet complete.")

    report_path = TASK_RESULTS[task_id].get("report_path", "")
    if not report_path or not os.path.exists(report_path):
        raise HTTPException(status_code=404, detail="Report file not found.")

    filename = os.path.basename(report_path)
    return FileResponse(
        report_path,
        media_type="text/markdown",
        headers={"Content-Disposition": f'attachment; filename="{filename}"'},
    )
```

// ─── Node chain rendering ─────────────────────────────────────────────────────
function buildNodeChain() {
  NODES.forEach(n => { nodeStates[n.key] = "pending"; });
  const list = document.getElementById("node-list");
  list.innerHTML = "";
  NODES.forEach((n, i) => {
    const item = document.createElement("div");

        <div class="node-icon" id="icon-${n.key}">

        <div class="node-text">

          <div class="node-status" id="status-${n.key}"></div>
        </div>
      </div>
      ${i < NODES.length - 1 ? `<div class="node-connector" id="conn-${n.key}"></div>` : ""}
    `;
    list.appendChild(item);
  });
}

function setNodeState(key, state) {
  nodeStates[key] = state;
  const icon   = document.getElementById(`icon-${key}`);
  const sym    = document.getElementById(`icon-sym-${key}`);
  const name   = document.getElementById(`name-${key}`);
  const status = document.getElementById(`status-${key}`);
  const conn   = document.getElementById(`conn-${key}`);
  if (!icon) return;

  icon.className   = `node-icon ${state}`;
  sym.textContent  = STATUS_ICON[state] || "○";
  name.className   = `node-name ${state}`;
  status.textContent = STATUS_TEXT[state] || "";

  if (conn) {
    conn.className = `node-connector ${state === "done" ? "done" : state === "running" ? "running" : ""}`;
  }

  // When a node finishes, auto-advance the NEXT pending node to running
  if (state === "done") {
    const idx = NODES.findIndex(n => n.key === key);
    for (let i = idx + 1; i < NODES.length; i++) {
      if (nodeStates[NODES[i].key] === "pending") {
        setNodeState(NODES[i].key, "running");
        break;
      }
    }
  }
}

// ─── Analysis ─────────────────────────────────────────────────────────────────
async function startAnalysis() {
  const prompt = document.getElementById("prompt-input").value.trim();
  if (!prompt || !selectedFile) return;

  showAnalysisPhase();

  const fd = new FormData();
  fd.append("file",   selectedFile);
  fd.append("prompt", prompt);
  fd.append("level",  selectedLevel);
  fd.append("tune",   tuneEnabled ? "true" : "false");

  let resp;
  try {
    resp = await fetch("/analyze", { method: "POST", body: fd });
  } catch (e) {
    showError("无法连接到后端服务，请确认 server.py 正在运行。");
    return;
  }
  if (!resp.ok) {
    const body = await resp.json().catch(() => ({}));
    showError(body.detail || `服务器错误 (${resp.status})`);
    return;
  }
  const { task_id } = await resp.json();
  currentTaskId = task_id;

  const es = new EventSource(`/progress/${task_id}`);
  es.onmessage = (e) => {
    const data = JSON.parse(e.data);
    if (data.node) {
      setNodeState(data.node, data.status);
    } else if (data.event === "complete") {
      es.close();
      showComplete(data);
    } else if (data.event === "error") {
      es.close();
      showError(data.message || "未知错误");
    }
  };
  es.onerror = () => {
    es.close();
    showError("与服务器的连接中断，请刷新页面重试。");
  };
}

function showComplete(data) {
  document.getElementById("download-pending").style.display = "none";
  const taskTypeMap = {
    regression:          "回归",
    classification:      "分类",
    clustering:          "聚类",
    anomaly_detection:   "异常检测",
    correlation_analysis:"相关性分析",
  };
  const taskLabel = taskTypeMap[data.task_type] || data.task_type || "未知";
  document.getElementById("complete-meta").textContent =
    `任务类型：${taskLabel} · 最优模型：${data.best_model || "N/A"} · 用时 ${data.elapsed} 秒`;
  document.getElementById("complete-card").classList.add("visible");
}

function showError(msg) {
  document.getElementById("download-pending").style.display = "none";
  document.getElementById("error-msg").textContent = msg;
  document.getElementById("error-card").classList.add("visible");
}

function downloadReport() {
  if (!currentTaskId) return;
  window.location.href = `/report/${currentTaskId}`;
}
